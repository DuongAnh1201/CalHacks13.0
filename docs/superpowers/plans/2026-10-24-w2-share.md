# W2 Share Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans (inline, no worktree, no push) to implement your task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The Share button starts a live Convex session on the sample case. A guest joins by link, then follows the host's MRI crosshair (Phase A), then the host's 3D page and mask (Phase B).

**Architecture:** The frontend uses `ConvexClient` (`convex/browser`) with untyped `anyApi` refs and the hand-written types in `share/types.ts`. A Zustand `useShare` store owns one `sessions.get` subscription. Phase A syncs voxel slice indices through Sage's existing `setPlane` / `setSlice` commands, with no Convex change. Phase B adds a `scene` field and new commands to `convex/`, and mirrors `/3d` state through `dispatch`.

**Tech Stack:**
- Vite 8, React 19, TS 6, react-router-dom 7, Zustand 5, `@niivue/niivue@0.69.0`
- vitest 5 (node env)
- Convex 1.48 (root), tested with vitest + `convex-test`

Spec: `docs/superpowers/specs/2026-10-24-w2-share-design.md`.

```ts
export const SESSION_ENDED = 'The host ended the session';
export const NO_SAMPLE = 'Sample scan is not in Convex. Run seed:seedSampleScan';
export const SYNC_FAILED = "Couldn't sync. Will try again on the next change.";
export const NOT_CONFIGURED = 'Sharing is not configured';
export const SAMPLE_ONLY = 'Sharing works on the sample case for now';

const COPY: Record<string, string> = {
  INVALID_INVITE: "That invite link doesn't work. Ask the host for a new one.",
  SESSION_ENDED,
};

type AppErrorLike = { data?: { code?: unknown; message?: unknown } } | null;

/** ConvexError.data.code from convex/lib/errors.ts, or null for anything else. */
export const codeOf = (e: unknown): string | null => {
  const c = (e as AppErrorLike)?.data?.code;
  return typeof c === 'string' ? c : null;
};

export function messageFor(e: unknown): string {
  const code = codeOf(e);
  if (code && code in COPY) return COPY[code];
  const m = (e as AppErrorLike)?.data?.message;
  return typeof m === 'string' ? m : 'Something went wrong. Try again.';
}
```

`frontend/src/share/stored.ts`:
```ts
export interface Stored { sessionId: string; participantToken: string; role: 'host' | 'viewer'; inviteCode: string }

type Store = Pick<Storage, 'getItem' | 'setItem' | 'removeItem'>;
const KEY = 'nv.share';

// sessionStorage, not localStorage: per tab, so host and guest can share one browser, and a reload keeps its token instead of joining twice.
function session(): Store | null {
  try { return globalThis.sessionStorage ?? null; } catch { return null; }
}

const isStored = (v: unknown): v is Stored => {
  const s = v as Stored | null;
  return !!s && typeof s.sessionId === 'string' && typeof s.participantToken === 'string' && (s.role === 'host' || s.role === 'viewer') && typeof s.inviteCode === 'string';
};

export function loadStored(s: Store | null = session()): Stored | null {
  try {
    const v: unknown = JSON.parse(s?.getItem(KEY) ?? 'null');
    return isStored(v) ? v : null;
  } catch { return null; }
}

export function saveStored(v: Stored, s: Store | null = session()): void {
  try { s?.setItem(KEY, JSON.stringify(v)); } catch { /* blocked: the session lasts until reload */ }
}

export function clearStored(s: Store | null = session()): void {
  try { s?.removeItem(KEY); } catch { /* blocked: nothing stored */ }
}
```

```ts
import type { ConvexClient } from 'convex/browser';
import { create } from 'zustand';
import { api, convex } from './client';
import { codeOf, messageFor, NO_SAMPLE, NOT_CONFIGURED, SESSION_ENDED, SYNC_FAILED } from './copy';
import { clearStored, loadStored, saveStored, type Stored } from './stored';
import type { SessionState, ViewerCommand } from './types';

type Client = Pick<ConvexClient, 'query' | 'mutation' | 'onUpdate'>;

interface ShareState { stored: Stored | null; session: SessionState | null; notice: string | null; syncNote: string | null }

let client: Client | null = convex;
let unsubscribe: (() => void) | null = null;

export const useShare = create<ShareState>()(() => ({ stored: loadStored(), session: null, notice: null, syncNote: null }));
export const shareConfigured = () => client !== null;
export function useClientForTest(c: Client | null): void { unsubscribe = null; client = c; }

// Tokens the server no longer accepts: forget them, no banner.
const STALE = new Set(['NOT_A_MEMBER', 'NOT_AUTHENTICATED', 'NOT_FOUND']);

const auth = (s: Stored) => ({ sessionId: s.sessionId, participantToken: s.participantToken });

function drop(notice: string | null): void {
  unsubscribe?.();
  unsubscribe = null;
  clearStored();
  useShare.setState({ stored: null, session: null, notice, syncNote: null });
}

function watch(s: Stored): void {
  unsubscribe?.();
  unsubscribe = client!.onUpdate(api.sessions.get, auth(s), (session: SessionState) => useShare.setState({ session }), (e: Error) => {
    const code = codeOf(e);
    if (code === 'SESSION_ENDED') drop(SESSION_ENDED);
    else if (code && STALE.has(code)) drop(null);
    else console.error('[share]', e);
  });
}

function begin(s: Stored): void {
  saveStored(s);
  useShare.setState({ stored: s, session: null, notice: null, syncNote: null });
  watch(s);
}

/** Re-subscribes with the token from this tab's storage, so a reload never joins twice. */
export function resume(): void {
  const s = useShare.getState().stored;
  if (client && s && !unsubscribe) watch(s);
}

// shortcut: a Convex deployment that is down makes these await until it is back, add a timeout if that bites in the demo.
export async function startShare(): Promise<string | null> {
  if (!client) return NOT_CONFIGURED;
  try {
    const scan = (await client.query(api.scans.getByJobId, { jobId: 'sample' })) as { scanId: string } | null;
    if (!scan) return NO_SAMPLE;
    const r = (await client.mutation(api.sessions.create, { scanId: scan.scanId, hostDisplayName: 'Host' })) as { sessionId: string; participantToken: string; inviteCode: string };
    begin({ sessionId: r.sessionId, participantToken: r.participantToken, role: 'host', inviteCode: r.inviteCode });
    return null;
  } catch (e) { return messageFor(e); }
}

export async function joinShare(code: string, name: string): Promise<string | null> {
  if (!client) return NOT_CONFIGURED;
  try {
    const r = (await client.mutation(api.sessions.join, { inviteCode: code, displayName: name })) as { sessionId: string; participantToken: string };
    begin({ sessionId: r.sessionId, participantToken: r.participantToken, role: 'viewer', inviteCode: code });
    return null;
  } catch (e) { return messageFor(e); }
}

export async function endShare(): Promise<void> {
  const s = useShare.getState().stored;
  if (!client || !s) return;
  drop(null); // unsubscribe first: otherwise the host's own subscription reports SESSION_ENDED
  try { await client.mutation(api.sessions.end, auth(s)); } catch (e) { console.error('[share]', e); }
}

export async function setFollowing(following: boolean): Promise<void> {
  const s = useShare.getState().stored;
  if (!client || !s) return;
  try { await client.mutation(api.sessions.setFollowMode, { ...auth(s), following }); } catch (e) { console.error('[share]', e); }
}

/** Host only. All mutations are issued before any await: the Convex client runs them in send order. */
export async function sendView(cmds: ViewerCommand[]): Promise<void> {
  const s = useShare.getState().stored;
  const c = client;
  if (!c || s?.role !== 'host' || cmds.length === 0) return;
  try {
    await Promise.all(cmds.map((command) => c.mutation(api.sessions.updateView, { ...auth(s), command })));
    if (useShare.getState().syncNote) useShare.setState({ syncNote: null });
  } catch (e) {
    console.error('[share]', e);
    useShare.setState({ syncNote: SYNC_FAILED });
  }
}
```

```ts
import type { Axis } from '../actions/types';
import { AXIS_INDEX, sliceOf } from '../viewer/slices';
import type { SliceIndices, ViewerCommand } from './types';

const AXES: readonly Axis[] = ['axial', 'coronal', 'sagittal'];

/** NiiVue crosshair fraction → 0-based voxel index; same rounding as the slice label. */
export const toIdx = (frac: number, n: number) => sliceOf(frac, n) - 1;
/** 0-based voxel index → fraction at the voxel centre. */
export const toFrac = (idx: number, n: number) => (idx + 0.5) / n;

/** pos is nv.scene.crosshairPos (RAS fractions), dimsRAS is nv.volumes[0].dimsRAS ([ndim, x, y, z]). */
export const indicesOf = (pos: ArrayLike<number>, dimsRAS: ArrayLike<number>): SliceIndices =>
  Object.fromEntries(AXES.map((a) => [a, toIdx(pos[AXIS_INDEX[a]], dimsRAS[AXIS_INDEX[a] + 1])])) as SliceIndices;

export const fracsOf = (idx: SliceIndices, dimsRAS: ArrayLike<number>): [number, number, number] =>
  [toFrac(idx.sagittal, dimsRAS[1]), toFrac(idx.coronal, dimsRAS[2]), toFrac(idx.axial, dimsRAS[3])];

export const sameIndices = (a: SliceIndices, b: SliceIndices) => AXES.every((x) => a[x] === b[x]);

/** Phase A: Sage's setSlice acts on the current plane, so each changed axis sends its plane first, every time. */
export function commandsFor(prev: SliceIndices, next: SliceIndices): ViewerCommand[] {
  return AXES.filter((a) => prev[a] !== next[a]).flatMap((a): ViewerCommand[] => [{ type: 'setPlane', plane: a }, { type: 'setSlice', index: next[a] }]);
}
```

```ts
import { useCallback, useRef } from 'react';
import { commandsFor, sameIndices } from './mprSync';
import { sendView, setFollowing, useShare } from './store';
import type { SliceIndices } from './types';

/** Phase A crosshair sync for /jobs/sample/mri. The host sends, a following guest receives. */
export function useMprShare(jobId: string): { onCrosshair: (idx: SliceIndices) => void; remote: SliceIndices | null } {
  const remote = useShare((s) => {
    const live = jobId === 'sample' && s.session?.status === 'active' && s.stored?.role === 'viewer' ? s.session : null;
    return live?.me.following ? live.view.sliceIndices : null;
  });
  // Last indices this host sent, per session: broadcastTo fires onLocationChange on all three views for one click.
  const sent = useRef<{ sessionId: string; idx: SliceIndices } | null>(null);

  const onCrosshair = useCallback((next: SliceIndices) => {
    const { stored, session } = useShare.getState();
    if (jobId !== 'sample' || session?.status !== 'active') return;
    if (stored?.role === 'host') {
      const prev = sent.current?.sessionId === session.sessionId ? sent.current.idx : session.view.sliceIndices;
      const cmds = commandsFor(prev, next);
      if (!cmds.length) return;
      sent.current = { sessionId: session.sessionId, idx: next };
      void sendView(cmds);
    } else if (session.me.following && !sameIndices(next, session.view.sliceIndices)) {
      void setFollowing(false); // a guest's own click: stop following and browse locally (spec §2)
    }
  }, [jobId]);

  return { onCrosshair, remote };
}
```

```ts
import type { Action, Hemi, Mode, ViewerState } from '../actions/types';
import type { Layer } from '../viewer/layers';
import type { ViewerCommand } from './types';

export type Page = 'mri' | '3d';
/** Mirror of convex/lib/viewCommands.ts Scene. */
export interface Scene {
  page: Page;
  mode: Mode;
  brainOpacity: number;
  explodeAmount: number;
  hidden: Record<Hemi, boolean>;
  layers: Record<Layer, boolean>;
  cameraTarget: 'brain' | 'tumor';
}

export const DEFAULT_SCENE: Scene = {
  page: 'mri', mode: 'whole', brainOpacity: 0.3, explodeAmount: 0,
  hidden: { left: false, right: false }, layers: { edema: true, core: true, enhancing: true }, cameraTarget: 'brain',
};

const SIDES: readonly Hemi[] = ['left', 'right'];
const LAYERS: readonly Layer[] = ['edema', 'core', 'enhancing'];

export const sceneOf = (s: ViewerState, page: Page): Scene => ({
  page, mode: s.mode, brainOpacity: s.brainOpacity, explodeAmount: s.explodeAmount, hidden: s.hidden, layers: s.layers, cameraTarget: s.cameraTarget,
});

export function sceneCommands(prev: Scene, next: Scene): ViewerCommand[] {
  const out: ViewerCommand[] = [];
  if (prev.page !== next.page) out.push({ type: 'setPage', page: next.page });
  if (prev.mode !== next.mode) out.push({ type: 'setMode', mode: next.mode });
  if (prev.brainOpacity !== next.brainOpacity) out.push({ type: 'setOpacity', value: next.brainOpacity });
  if (prev.explodeAmount !== next.explodeAmount) out.push({ type: 'setExplosion', value: next.explodeAmount });
  for (const side of SIDES) if (prev.hidden[side] !== next.hidden[side]) out.push({ type: 'setHemisphereVisible', side, visible: !next.hidden[side] });
  for (const layer of LAYERS) if (prev.layers[layer] !== next.layers[layer]) out.push({ type: 'setLayerVisible', layer, visible: next.layers[layer] });
  if (prev.cameraTarget !== next.cameraTarget) out.push({ type: 'setCameraTarget', target: next.cameraTarget });
  return out;
}

/** Guest side. prev null = apply everything. A brain camera dispatches nothing: the guest's own orbit stays put. */
export function sceneActions(prev: Scene | null, next: Scene): Action[] {
  const out: Action[] = [];
  if (next.cameraTarget === 'tumor' && prev?.cameraTarget !== 'tumor') out.push({ type: 'focusTumor' });
  if (!prev || prev.mode !== next.mode) out.push({ type: 'setMode', mode: next.mode });
  if (!prev || prev.brainOpacity !== next.brainOpacity) out.push({ type: 'setOpacity', value: next.brainOpacity });
  if (!prev || prev.explodeAmount !== next.explodeAmount) out.push({ type: 'setExplosion', value: next.explodeAmount });
  for (const side of SIDES) if (!prev || prev.hidden[side] !== next.hidden[side]) out.push({ type: 'setHemisphereVisible', side, visible: !next.hidden[side] });
  for (const layer of LAYERS) if (!prev || prev.layers[layer] !== next.layers[layer]) out.push({ type: 'setLayerVisible', layer, visible: next.layers[layer] });
  return out;
}

/** Host side: one trailing send per `ms` with whatever changed since the last send (spec §3 throttle, applied to all fields). */
export function sceneSender(base: Scene, send: (cmds: ViewerCommand[]) => void, ms = 150) {
  let sent = base;
  let latest = base;
  let timer: ReturnType<typeof setTimeout> | null = null;
  const flush = () => {
    if (timer) { clearTimeout(timer); timer = null; }
    const cmds = sceneCommands(sent, latest);
    sent = latest;
    if (cmds.length) send(cmds);
  };
  return {
    push(s: Scene) { latest = s; timer ??= setTimeout(flush, ms); },
    flush,
  };
}
```
