# Live Group Call Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:executing-plans, inline, with the overrides in "Worker rules" below. Steps use checkbox (`- [ ]`) syntax. Tick your own boxes in this file when a step is done, and commit the ticks with your task.

**Goal:** Members of a live session join a voice call from the top bar, hear each other, see a ring on whoever is speaking (face pile + 3D cursor chip), and, while unmuted with the host's transcript switch on, their speech feeds the existing live transcript.

**Architecture:**
- **Convex:** `convex/call.ts` holds a `member` internalQuery (wraps `requireSessionMember`) and a public `token` action. The action mints a LiveKit JWT with `livekit-server-sdk` (`AccessToken`). Room = `sessionId`, identity = `participantId`. Audio never touches Convex.
- **Frontend data:** `share/store.ts` gains three thin senders: `callToken`, `sendTranscript` and `watchTranscriptEnabled`. `frontend/src/call/store.ts` is a zustand store plus module-scope functions (`join`, `leave`, `toggleMute`). It holds the LiveKit `Room`, which is loaded with a dynamic `import('livekit-client')` on first Join.
- **UI:** `call/CallControls.tsx` is rendered by `share/ShareControls.tsx`. `FacePile` and `RemoteCursors` read `useCall.speaking` for the ring.
- **Transcript:** `call/transcribeLink.ts` starts and stops #135's `createTranscriber` from the call state.

**Tech Stack:**
- Convex 1.48, convex-test + vitest (edge-runtime), `livekit-server-sdk@2.19.1`.
- React 19, zustand 5, `livekit-client@2.22.4`.
- Frontend vitest uses the node env, runs `src/**/*.test.ts` only, and has no DOM.

Spec: `docs/superpowers/specs/2026-10-24-live-call-design.md`. Branch: `viewer/live-call` (rebased on `origin/main` 0a78572, which includes #135/#136 transcript and #140 workspace UI).

## Global Constraints

- **Auth:** the call token is minted only after `requireSessionMember` (`convex/lib/auth.ts`). Identity and name come from the participant row, never from args.
- **Errors:** throw with `appError(code, message)` (`convex/lib/errors.ts`). Missing LiveKit env → `appError("UNAVAILABLE", "Call unavailable")`.
- **Grant:**
  - Exactly `{ roomJoin: true, room: sessionId, canPublish: true, canPublishSources: [TrackSource.MICROPHONE], canSubscribe: true, canPublishData: false }`.
  - Never `roomAdmin`, `roomCreate`, `roomList` or `roomRecord`.
- **Token TTL:** 900 s.
- **Env:** `LIVEKIT_API_KEY`, `LIVEKIT_API_SECRET`, `LIVEKIT_URL`. Read through `env()` (`convex/lib/env.ts`). Never log them, return the key or secret, or commit them. `LIVEKIT_URL` (public `wss://`) is returned to the client.
- **No Convex writes at audio or speaking rate.** Speaking state is client-only.
- **Frontend Convex types:** no codegen. Use `anyApi` (`share/client.ts`) and hand-written types.
- **Routes:** call UI shows only on `/jobs/<id>/(mri|3d)` while `session.status === 'active'` (same regex as `FacePile`).
- **Copy:** `Join call`, `Joining…`, `Mute`, `Unmute`, `Leave`, `Microphone blocked`, `Couldn't connect to the call`, `Call dropped. Join again`.
- **Privacy:** audio goes through LiveKit Cloud. De-identified sample case only; no recording or egress.
- **Out of scope** (do not touch):
  - `convex/transcript.ts`, `frontend/src/agent/transcribe.ts` (use as is).
  - The transcript tab and banner (#139), host call controls, video, LiveKit webhooks, the `convex-livekit` component.

```ts
/**
 * Live group call (LiveKit Cloud). Convex only authorizes the caller and mints
 * the room join token; audio and speaking state never touch Convex.
 * Room = sessionId, identity = participantId. Spec: docs/superpowers/specs/2026-10-24-live-call-design.md.
 */

import { v } from "convex/values";
import { AccessToken, TrackSource } from "livekit-server-sdk";
import { internal } from "./_generated/api";
import { action, internalQuery } from "./_generated/server";
import { requireSessionMember } from "./lib/auth";
import { env } from "./lib/env";
import { appError } from "./lib/errors";

const TOKEN_TTL_S = 15 * 60;

const vCaller = { sessionId: v.id("sessions"), participantToken: v.string() };

/** The member check behind `token`; actions have no db. */
export const member = internalQuery({
  args: vCaller,
  handler: async (ctx, args): Promise<{ participantId: string; displayName: string }> => {
    const { participant } = await requireSessionMember(ctx, args);
    return { participantId: participant._id, displayName: participant.displayName };
  },
});

/** A mic-only join token for this session's room. */
export const token = action({
  args: vCaller,
  handler: async (ctx, args): Promise<{ url: string; token: string }> => {
    const apiKey = env("LIVEKIT_API_KEY");
    const apiSecret = env("LIVEKIT_API_SECRET");
    const url = env("LIVEKIT_URL");
    if (!apiKey || !apiSecret || !url) throw appError("UNAVAILABLE", "Call unavailable");

    const me = await ctx.runQuery(internal.call.member, args);
    const at = new AccessToken(apiKey, apiSecret, { identity: me.participantId, name: me.displayName, ttl: TOKEN_TTL_S });
    at.addGrant({
      roomJoin: true,
      room: args.sessionId,
      canPublish: true,
      canPublishSources: [TrackSource.MICROPHONE],
      canSubscribe: true,
      canPublishData: false,
    });
    return { url, token: await at.toJwt() };
  },
});
```

```ts
export async function callToken(): Promise<{ url: string; token: string } | { error: string }> {
  const s = useShare.getState().stored;
  if (!client || !s) return { error: NOT_CONFIGURED };
  try { return await client.action(api.call.token, auth(s)); } catch (e) { return { error: messageFor(e) }; }
```

```ts
import { create } from 'zustand';
import { callToken, useShare } from '../share/store';

export type CallStatus = 'idle' | 'connecting' | 'live' | 'error';
interface CallState { status: CallStatus; muted: boolean; speaking: Set<string>; error: string | null }

type CallTrack = { kind: string; attach(): HTMLMediaElement; detach(): HTMLMediaElement[] };
/** The slice of livekit-client's Room this store uses; tests pass a fake. */
export interface CallRoom {
  on(e: 'trackSubscribed' | 'trackUnsubscribed', fn: (t: CallTrack) => void): unknown;
  on(e: 'activeSpeakersChanged', fn: (speakers: { identity: string }[]) => void): unknown;
  on(e: 'disconnected', fn: () => void): unknown;
  connect(url: string, token: string): Promise<void>;
  disconnect(): Promise<void>;
  startAudio(): Promise<void>;
  localParticipant: { identity: string; setMicrophoneEnabled(on: boolean): Promise<unknown> };
}
type RoomFactory = () => Promise<CallRoom>;

export const MIC_BLOCKED = 'Microphone blocked';
export const CONNECT_FAILED = "Couldn't connect to the call";
export const CALL_DROPPED = 'Call dropped. Join again';

// Dynamic import: livekit-client loads on first Join only, and never in node tests.
const liveKitRoom: RoomFactory = async () => { const { Room } = await import('livekit-client'); return new Room() as unknown as CallRoom; };
let makeRoom = liveKitRoom;
let room: CallRoom | null = null;
let stopWatch: (() => void) | null = null;
const audio = new Set<HTMLMediaElement>();

export const useCall = create<CallState>()(() => ({ status: 'idle', muted: false, speaking: new Set<string>(), error: null }));
export function useRoomForTest(f: RoomFactory | null): void { makeRoom = f ?? liveKitRoom; room = null; }

function cleanup(): void {
  room = null;
  stopWatch?.();
  stopWatch = null;
  for (const el of audio) el.remove();
  audio.clear();
  useCall.setState({ speaking: new Set() });
}

async function fail(r: CallRoom, error: string): Promise<void> {
  cleanup();
  useCall.setState({ status: 'error', error });
  await r.disconnect().catch(() => {});
}

/** Connects this tab to the session's room with the mic live. Call from a click: it unlocks audio playback. */
export async function join(): Promise<void> {
  const { status } = useCall.getState();
  if (status === 'connecting' || status === 'live') return;
  useCall.setState({ status: 'connecting', error: null, muted: false });
  const t = await callToken();
  if (useCall.getState().status !== 'connecting') return; // left while fetching the token
  if ('error' in t) { useCall.setState({ status: 'error', error: t.error }); return; }

  const r = await makeRoom();
  if (useCall.getState().status !== 'connecting') return;
  room = r;
  r.on('trackSubscribed', (track) => {
    if (track.kind !== 'audio') return;
    const el = track.attach();
    el.hidden = true;
    document.body.append(el);
    audio.add(el);
  });
  r.on('trackUnsubscribed', (track) => { for (const el of track.detach()) { el.remove(); audio.delete(el); } });
  r.on('activeSpeakersChanged', (speakers) => useCall.setState({ speaking: new Set(speakers.map((p) => p.identity)) }));
  r.on('disconnected', () => { if (room === r) { cleanup(); useCall.setState({ status: 'idle', error: CALL_DROPPED }); } });
  // Leaving or losing the share session ends the call.
  stopWatch = useShare.subscribe((s) => { if (!s.stored || s.session?.status === 'ended') void leave(); });

  try { await r.connect(t.url, t.token); await r.startAudio(); } catch { if (room === r) await fail(r, CONNECT_FAILED); return; }
  if (room !== r) return; // left while connecting
  try { await r.localParticipant.setMicrophoneEnabled(true); } catch { if (room === r) await fail(r, MIC_BLOCKED); return; }
  if (room === r) useCall.setState({ status: 'live' });
}

export async function leave(): Promise<void> {
  const r = room;
  cleanup();
  useCall.setState({ status: 'idle', muted: false, error: null });
  await r?.disconnect();
}

export async function toggleMute(): Promise<void> {
  const r = room;
  if (!r || useCall.getState().status !== 'live') return;
  const muted = !useCall.getState().muted;
  await r.localParticipant.setMicrophoneEnabled(!muted);
  useCall.setState((s) => {
    const speaking = new Set(s.speaking);
    if (muted) speaking.delete(r.localParticipant.identity);
    return { muted, speaking };
  });
}
```

```ts
export async function sendTranscript(text: string, durationMs: number): Promise<void> {
  const s = useShare.getState().stored;
  if (!client || !s) return;
  try { await client.mutation(api.transcript.append, { ...auth(s), text, durationMs }); } catch (e) { console.error('[transcript]', e); }
}

/** The host's transcript switch for this session. Returns an unsubscribe; reports false on error. */
export function watchTranscriptEnabled(onEnabled: (on: boolean) => void): () => void {
  const s = useShare.getState().stored;
  if (!client || !s) return () => {};
  return client.onUpdate(api.transcript.state, auth(s), (r: { enabled: boolean }) => onEnabled(r.enabled), () => onEnabled(false));
}
```

```ts
import { createTranscriber, type Transcriber, type TranscriptSegment } from '../agent/transcribe';
import { sendTranscript, useShare, watchTranscriptEnabled } from '../share/store';
import { useCall } from './store';

type Make = (onSegment: (s: TranscriptSegment) => void) => Transcriber;

/**
 * Speech-to-text on this tab's mic while in the call, unmuted, and with the host's
 * transcript switch on (#135). Returns an unlink.
 */
export function linkTranscriber(make: Make = (onSegment) => createTranscriber({ onSegment })): () => void {
  const t = make((s) => void sendTranscript(s.text, s.endedAt - s.startedAt));
  let enabled = false;
  let watching: string | null = null; // sessionId the switch subscription is for
  let unwatch = () => {};

  const sync = () => {
    const { status, muted } = useCall.getState();
    const key = status === 'live' ? useShare.getState().stored?.sessionId ?? null : null;
    if (key !== watching) {
      unwatch();
      enabled = false;
      watching = key;
      unwatch = key ? watchTranscriptEnabled((on) => { enabled = on; sync(); }) : () => {};
    }
    const want = t.supported && status === 'live' && !muted && enabled;
    if (want && !t.running()) t.start();
    else if (!want && t.running()) t.stop();
  };

  const offCall = useCall.subscribe(sync);
  const offShare = useShare.subscribe(sync);
  sync();
  return () => { offCall(); offShare(); unwatch(); if (t.running()) t.stop(); };
}
```
